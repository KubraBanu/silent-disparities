# --- 2. Gender Fairness Metric Function ---
def compute_gender_metrics(pred_df, model_name):
    """
    Computes accuracy and avg predicted probability by gender.
    Works on serverless environments by avoiding RDD operations
    where possible (though .collect() is still used on the filtered aggregation).
    """

    print(f"\n=== Gender-specific metrics for {model_name} ===")

    # Ensure 'prob1' is available, which is the probability of the positive class (1).
    # This is critical for measuring the average predicted risk.
    pred_df = pred_df.withColumn(
        "prob1",
        vector_to_array("probability")[1]
    )

    # Get list of unique genders (e.g., ['M', 'F', 'OTHER'])
    # Note: .collect() here is acceptable since the number of genders is small.

        sub = df.filter(col("gender") == g)

    for g, total, acc, avg_prob in metrics:
        print(f"Gender: {g} | n={total} | Accuracy={acc:.4f} | Avg Prob={avg_prob:.4f}")

    return metrics

temp_model = dt_temp.fit(train_hashed)

importances = temp_model.featureImportances.toArray()

# Select top 64 hashed feature dimensions (indices)
# importances.argsort() gives indices sorted by score. [-64:] selects the last 64 (the highest scores).
selected_dims = [int(x) for x in importances.argsort()[-64:]]

print(f"Selected top {len(selected_dims)} hashed feature dimensions.")

# --------------------------------------------------------------
# STEP 2 — UDF to retain only top-K hashed feature dimensions
# --------------------------------------------------------------

@udf(returnType=VectorUDT())
def keep_top_k(v):
    """
    Creates a reduced dense vector containing only the selected top-K feature slots.
    Note: Using VectorSlicer (PySpark ML Transformer) is generally recommended
    over UDFs for better performance.
    """
    # SparseVector → extract via indexing
    if isinstance(v, SparseVector):
        return DenseVector([float(v[i]) for i in selected_dims])

    # DenseVector → direct indexing
    if isinstance(v, DenseVector):
        return DenseVector([float(v[i]) for i in selected_dims])

    # fallback (should not happen)
    return DenseVector([0.0] * len(selected_dims))

# STEP 3 — Apply top-K reduction to training and test data
# --------------------------------------------------------------

# Note: The 'hashed_features' must be generated before applying the UDF.

# This function prepares and plots the ROC curve for multiple models.

def plot_roc_for_models(models, title):
    """
    Plots the Receiver Operating Characteristic (ROC) curve for a list of models.

    models: list of tuples (label_name, pandas_dataframe_with_label_and_prob1)
    title:  title for the ROC figure
    """
    plt.figure(figsize=(8, 6))

    for model_name, pdf in models:
        # pdf must contain 'label' (0/1) and 'prob1' (predicted probability of class 1)
        y_true = pdf["label"].values
        y_score = pdf["prob1"].values

        # Compute ROC curve and AUC
        fpr, tpr, _ = roc_curve(y_true, y_score)
        roc_auc = auc(fpr, tpr)

        # Plot the model's ROC curve
        plt.plot(fpr, tpr, linewidth=2, label=f"{model_name} (AUC = {roc_auc:.3f})")

    # Diagonal reference line (AUC=0.5, random classifier)
    plt.plot([0, 1], [0, 1], linestyle="--", linewidth=1, color='gray', label='Random (AUC = 0.500)')

    plt.xlabel("False Positive Rate (1 - Specificity)")
    plt.ylabel("True Positive Rate (Sensitivity)")
    plt.title(title)
    plt.legend(loc="lower right")
    plt.grid(alpha=0.3)
    plt.tight_layout()
    plt.show()

print("ROC plotting helper function 'plot_roc_for_models' defined.")
